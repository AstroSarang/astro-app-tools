function norm360(x){ return ((x % 360) + 360) % 360; }

function parseAstroCode(code) {
  const m = code.trim().match(/^AS1-(\d{8})-(\d{6})-([+-]\d{4})-(\d{4})([NS])-(\d{5})([EW])$/i);
  if(!m) return null;
  return {
    yy: +m[1].slice(0,4), mm: +m[1].slice(4,6), dd: +m[1].slice(6,8),
    hh: +m[2].slice(0,2), mi: +m[2].slice(2,4), ss: +m[2].slice(4,6),
    tz: m[3].slice(0,3) + ':' + m[3].slice(3),
    lat: (+m[4].slice(0,2) + (+m[4].slice(2))/60) * (m[5].toUpperCase()==='S' ? -1 : 1),
    lon: (+m[6].slice(0,3) + (+m[6].slice(3))/60) * (m[7].toUpperCase()==='W' ? -1 : 1)
  };
}

function toJulian(yy, mm, dd, hh, mi, ss, tzStr) {
  const m = tzStr.match(/([+-])(\d+):(\d+)/);
  const tzH = (+m[2] + (+m[3])/60) * (m[1]==='-' ? -1 : 1);
  const utcH = hh + mi/60 + ss/3600 - tzH;
  let y = yy, mo = mm;
  if(mo <= 2){ y--; mo += 12; }
  const A = Math.floor(y/100), B = 2 - A + Math.floor(A/4);
  return Math.floor(365.25*(y+4716)) + Math.floor(30.6001*(mo+1)) + dd + B - 1524.5 + utcH/24;
}

function getSubLords(nakIdx) {
  const total = 800.0, start = nakIdx % 9;
  const res = []; let pos = 0;
  for(let i=0; i<9; i++){
    const lord = VIM_ORDER[(start+i)%9];
    const w = VIM_YEARS[lord]/120*total;
    res.push({lord, s:pos, e:pos+w}); pos += w;
  }
  return res;
}

function getLords(absMin, nakIdx) {
  const nakStart = nakIdx * 800;
  const din = absMin - nakStart;
  const subs = getSubLords(nakIdx);
  for(const sub of subs){
    if(sub.s <= din && din < sub.e){
      const si = VIM_ORDER.indexOf(sub.lord);
      const wtot = sub.e - sub.s;
      let pos = sub.s;
      for(let i=0; i<9; i++){
        const ssl = VIM_ORDER[(si+i)%9];
        const w = VIM_YEARS[ssl]/120*wtot;
        if(pos <= din && din < pos+w) return {sl: sub.lord, ssl};
        pos += w;
      }
    }
  }
  return {sl:'?', ssl:'?'};
}

function planetLords(lon) {
  const rashi = Math.floor(norm360(lon)/30)%12;
  const nakIdx = Math.floor(norm360(lon)/(360/27))%27;
  const absMin = norm360(lon)*60;
  const {sl, ssl} = getLords(absMin, nakIdx);
  return {rashi, nakIdx, nl: NAK_LORDS[nakIdx], sl, ssl, pada: Math.floor((absMin - nakIdx*800)/200)+1};
}

function nakLord(lon){ return NAK_LORDS[Math.floor(norm360(lon)/(360/27))%27]; }
function d9Sign(lon){ return Math.floor(norm360(lon)/(30/9))%12; } // textbook D9 shortcut (verified identical)

function aspectsOf(planet, sign){
  const s = new Set([(sign+6)%12]);
  if(planet==='Ma'){ s.add((sign+3)%12); s.add((sign+7)%12); }
  if(planet==='Ju'){ s.add((sign+4)%12); s.add((sign+8)%12); }
  if(planet==='Sa'){ s.add((sign+2)%12); s.add((sign+9)%12); }
  if(planet==='Ra' || planet==='Ke'){ s.add((sign+4)%12); s.add((sign+8)%12); }
  return s;
}

function arudha(houseSign, lagna){
  let a = (2*houseSign - lagna) % 12;
  a = ((a % 12) + 12) % 12;
  if(a === lagna || a === (lagna+6)%12) a = (a+10)%12;
  return a;
}

function buildChart(p){
  const jd = toJulian(p.yy, p.mm, p.dd, p.hh, p.mi, p.ss, p.tz);
  swe.setSiderealMode(SID_MODE_LAHIRI);
  const lahiriAya = swe.getAyanamsa(jd);
  const kpAya = lahiriAya - KP_AYA_OFFSET/60;
  const BODIES = [SEP.Sun, SEP.Moon, SEP.Mars, SEP.Mercury, SEP.Jupiter, SEP.Venus, SEP.Saturn, SLP.MeanNode];
  const trop = {};
  for(let i=0; i<8; i++){
    const pos = swe.calculatePosition(jd, BODIES[i]);
    trop[PLANETS[i]] = pos.longitude;
  }
  const sid = {}, sidKP = {};
  for(const pl of ['Su','Mo','Ma','Me','Ju','Ve','Sa','Ra']){
    sid[pl] = norm360(trop[pl] - lahiriAya);
    sidKP[pl] = norm360(trop[pl] - kpAya);
  }
  sid['Ke'] = norm360(sid['Ra'] + 180);
  sidKP['Ke'] = norm360(sidKP['Ra'] + 180);
  // Placidus cusps (tropical -> sidereal), KP tool rotation check
  const hs = swe.calculateHouses(jd, p.lat, p.lon, 'P');
  let housesArr = (hs && (hs.cusps || hs.houses)) || [];
  if(housesArr.length < 12) throw new Error('House data invalid: ' + housesArr.length + ' cusps');
  const ascCheck = norm360(housesArr[1] - kpAya);
  const ascSidKP = norm360(hs.ascendant - kpAya);
  let rotated = housesArr;
  if(Math.abs(ascCheck - ascSidKP) < 1.0) rotated = housesArr.slice(1).concat([housesArr[0]]);
  const cuspsLahiri = [], cuspsKP = [];
  for(let i=0; i<12; i++){
    cuspsLahiri.push(norm360(rotated[i] - lahiriAya));
    cuspsKP.push(norm360(rotated[i] - kpAya));
  }
  const ascSid = norm360(hs.ascendant - lahiriAya);
  const lagna = Math.floor(ascSid/30)%12;
  const psign = {}, pdeg = {};
  for(const pl of PLANETS){ psign[pl] = Math.floor(sid[pl]/30)%12; pdeg[pl] = sid[pl]%30; }
  const d9 = {};
  for(const pl of PLANETS) d9[pl] = d9Sign(sid[pl]);
  const d9Lagna = d9Sign(ascSid);
  const d9LagnaLon = d9Lagna*30 + ((ascSid%30) % (30/9))*9;
  // KP: 7th cusp sub-lord and its lords
  const csl = planetLords(cuspsKP[6]).sl;
  const cslLon = sidKP[csl];
  const cslLords = planetLords(cslLon);
  // Jaimini karakas (7 planets, degrees within sign)
  const seven = ['Su','Mo','Ma','Me','Ju','Ve','Sa'];
  let ak = seven[0], dk = seven[0];
  for(const pl of seven){ if(pdeg[pl] > pdeg[ak]) ak = pl; if(pdeg[pl] < pdeg[dk]) dk = pl; }
  const swamsa = psign[ak], karakamsa = d9[ak];
  const ul = arudha((lagna+11)%12, lagna);
  const a7 = arudha((lagna+6)%12, lagna);
  return {lagna, ascSid, sid, sidKP, psign, pdeg, d9, d9Lagna, d9LagnaLon,
    cuspsLahiri, cuspsKP, csl, cslSl: cslLords.sl, cslNl: cslLords.nl, cslSsl: cslLords.ssl,
    ak, dk, swamsa, karakamsa, ul, a7};
}

function countPlanets(items, sections){
  const cnt = {};
  for(const pl of PLANETS) cnt[pl] = 0;
  for(const s of sections){
    for(const it of items[s]) for(const pl of it[2]) cnt[pl]++;
  }
  return cnt;
}

function top1Of(cnt){
  let mx = -1;
  for(const pl of PLANETS) if(cnt[pl] > mx) mx = cnt[pl];
  const tops = PLANETS.filter(pl => cnt[pl] === mx);
  let owning = tops.filter(pl => OWNED[pl].length > 0);
  if(owning.length === 0){
    // top-1 was Rahu/Ketu only: fall back to next most frequent sign-owning planet
    const rest = PLANETS.filter(pl => OWNED[pl].length > 0).sort((a,b) => cnt[b]-cnt[a]);
    owning = rest.length ? [rest[0]] : [];
  }
  const signs = [...new Set(owning.flatMap(pl => OWNED[pl]))].sort((a,b) => a-b);
  return {tops, owning, mx, signs};
}

function mostFreq(xs){
  const c = new Map();
  for(const x of xs) c.set(x, (c.get(x)||0)+1);
  const mx = Math.max(...c.values());
  return {winners: [...c.keys()].filter(x => c.get(x)===mx), mx};
}

function r1Set(lord7Sign){
  // तत्त्व की 3 राशियाँ + हर राशि के स्वामी की दूसरी राशि (सूर्य/चंद्र की एक ही राशि है)
  const base = ELEM_SIGNS[ELEMENT[lord7Sign]];
  const extra = [];
  for(const s of base){
    const lord = SIGN_LORD[s];
    for(const os of OWNED[lord]){
      if(os !== s && !base.includes(os) && !extra.includes(os)) extra.push(os);
    }
  }
  return base.concat(extra);
}

  const gender = document.getElementById('gender').value;
  try {
